#     libbpf-dev \

#     linux-tools-generic \
#     bpfcc-tools \
#     python3-bpfcc \
#     iproute2 \
#     iperf3 \
#     hping3 \
#     mosquitto \
#     mosquitto-clients \
#     nginx \
#     tcpdump \
#     tshark \
#     jq \

#     make \
#     python3-pip \
#     python3-venv

# Note on BCC:
#   python3-bpfcc is installed from system apt, not pip.
